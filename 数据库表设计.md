# 数据库表设计

## 用户表（users）

```sql
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,        -- 用户名
    password VARCHAR(255) NOT NULL,              -- 密码
    phone VARCHAR(20) UNIQUE NOT NULL,           -- 电话
    email VARCHAR(100) UNIQUE,                   -- 邮箱
    full_name VARCHAR(100) NOT NULL,             -- 姓名
    status TINYINT DEFAULT 1,                    -- 状态（0 - 禁用，1 - 启用）
    role TINYINT DEFAULT 1,                      -- 身份（0 - 管理员，1 - 技术人员）
    department VARCHAR(100),                     -- 部门
    created_time DATETIME,                       -- 创建时间（入职时间）
    last_login DATETIME                          -- 最后一次登录时间
);
```



## 文档表（documents）

```sql
CREATE TABLE documents (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,                  -- 文档标题
    contributor_id INT,                           -- 贡献者id
    first_edit_date DATETIME,                     -- 首次编辑时间

    problem_intro TEXT NOT NULL,                  -- 问题简介（含现象、频率、后果）
    image_urls TEXT,                              -- 图片url（逗号分隔）
    causes TEXT NOT NULL,                         -- 原因（高频到低频）
    evaluation TEXT NOT NULL,                     -- 评估（手段、方法、工具）
    inspection TEXT NOT NULL,                     -- 检查（现场定位确认）
    solutions TEXT NOT NULL,                      -- 解决方案（临时+根本）
    key_points TEXT NOT NULL                      -- 关键点（原因、后果、措施、复盘）
);

alter table documents
    add is_vectorized tinyint default 0 not null;

alter table documents
    add vector_update_time datetime null;
```

## 消息（message）

```sql
CREATE TABLE message (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    session_id INT NOT NULL,              -- 对应的对话id 
    message_order INT NOT NULL,           -- 在对话中的轮数
    role TINYINT NOT NULL,                -- 发送方，0 - ai，1 - 用户
    content_text LONGTEXT,                -- 消息内容
    user_uploaded_images TEXT,            -- 用户上传的图片url（逗号分隔）
    ai_reference_doc_ids TEXT,            -- ai最终回答关联的文档编号
    created_time DATETIME,                -- 创建时间

    INDEX idx_session_order (session_id, message_order),
    FOREIGN KEY (session_id) REFERENCES conversation(id)
);
```



## 对话表（conversation）

```sql
CREATE TABLE conversation(
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,                 -- 发起对话的用户id
    title VARCHAR(255),                   -- 本次对话的标题
    created_time DATETIME,                -- 创建时间
    updated_time DATETIME,                -- 最后一条消息发送时间
    INDEX idx_user_id (user_id),
    FOREIGN KEY (user_id) REFERENCES users(id)
);
```
